% Source of truth: docs/ARCHITECTURE.md §9 (rules R1–R7). Needs \input{figures/tikz-setup}.
% Size: about 15.6 cm x 3.7 cm.
\begin{tikzpicture}[x=1cm, y=1cm, every node/.append style={}]
  % ---------------------------------------------------------------- states
  \tikzset{sr/state/.style={minimum height=8mm}}
  \node[sr/base, sr/state, fill=white, minimum width=2.55cm] (sel) at (1.275,1.95) {Select acceptable\\sessions + preference};
  \node[sr/client, sr/state, minimum width=2.10cm] (que) at (5.05,1.95)  {\textbf{Queued}};
  \node[sr/client, sr/state, minimum width=2.30cm] (off) at (9.10,1.95)  {\textbf{Offered}\\hold: 15\,min by default,\\never past the cut-off};
  \node[sr/base, sr/state, fill=srGrid!45, minimum width=2.30cm] (con) at (13.55,1.95) {\textbf{Confirmed}};
  \node[sr/base, fill=srGrid!45, minimum width=10.20cm, minimum height=0.50cm] (can) at (9.30,3.00)
    {\textbf{Cancelled} \textemdash{} leaves the whole workshop; re-registering re-enters the queue at the back};

  % ---------------------------------------------------------------- transitions
  \draw[sr/flow] (sel) -- (que);
  \draw[sr/flow] (que) -- node[sr/lbl, above=1pt] {seat free} node[sr/lbl, below=1pt] {by preference} (off);
  \draw[sr/flow] (off) -- node[sr/lbl, above=1pt] {accept} (con);
  % cancel from any live state (R6)
  \draw[sr/async] (que.north) -- (que.north |- can.south);
  \draw[sr/async] (off.north) -- (off.north |- can.south);
  \draw[sr/async] (con.north) -- (con.north |- can.south);
  % decline or expire (R5)
  \draw[sr/flow] ($(off.south)+(0.35,0)$) -- ++(0,-0.28) -| node[sr/lbl, pos=0.25, below=1pt]
    {decline or expire: drop that session only, keep the rest;\\with none left the registration ends} ($(que.south)+(0.55,0)$);
  % backfill: released seat returns to the queue (dashed blue)
  \draw[sr/flow, draw=srBlue, dash pattern=on 2.4pt off 1.4pt] ($(con.south)$) -- ++(0,-1.20) coordinate (bfr) -|
    ($(que.south)+(-0.80,0)$);
  \node[sr/lbl, text=srBlue, anchor=south east] at ($(bfr)+(-0.12,0.02)$) {released seat returns to the queue (backfill)};

  % ---------------------------------------------------------------- invariant callout
  \node[sr/base, fill=srYellow!10, dash pattern=on 2pt off 1.2pt, text width=15.3cm, minimum height=0.5cm,
        anchor=north west, inner ysep=1.5pt, align=left] (inv) at (0,0.18)
    {\textbf{Invariants enforced inside the allocation transaction:} one verified account holds at most \emph{one}
     pending hold or confirmed seat per workshop \enspace$\bullet$\enspace confirmed + unreleased holds $\leq$ session
     capacity \enspace$\bullet$\enspace no new allocation within 30\,min of session start};
\end{tikzpicture}
