\begin{frame}{Problem: one form per session wastes seats}
  \begin{itemize}
    \item Students book two sessions ``to be safe'': one person, two seats
    \item A dropout's seat stays empty unless someone e-mails the waitlist
    \item \hl{SeatRelay:} one registration, ranked sessions, one queue, a timed hold
  \end{itemize}
  \vspace{0.40in}
  \stat{$\geq$\,\MarketStudentOrgs}{student organisations\\at five universities}\hfill
  \stat{\MarketNusLibcalRepeatEvents{} of \MarketNusLibcalWorkshops}{NUS Libraries workshop\\events were repeats}\hfill
  \stat{\sd\MarketTamCampusSgd\,$\cdot\,p$}{campus market a year;\\$p$: \IfPending{\InterviewPEventWeightedPct}{\pending{interviews}}{\InterviewPEventWeightedPct\,\%}}
  \source{Bottom-up sizing, lower bounds: market = organisations $\times$ $p$ (share with multi-session
    sign-ups) $\times$ \sd\MarketPriceSgd{} a year. NUS Libraries calendar \MarketLibcalFrom{} to
    \MarketLibcalTo. Final report \S1, Table 1; \code{results/market/}.}
\end{frame}

\begin{frame}{Business model: organisers pay, participants never do}
  \begin{itemize}
    \setlength{\itemsep}{0.42em}
    \item Free tier: one open workshop \textperiodcentered{} organiser plan: \hl{\sd10 a month} (\sd120 a year)
    \item Worth it if hours saved per event $\times$ value of an hour $\times$ events a year $\geq$ \sd120
    \item Hours saved and willingness to pay: \IfPending{\InterviewHoursMedian}{\pending{organiser interviews}}{median \InterviewHoursMedian{} h per event; bargain at \sd\InterviewWtpBargainMedian/month}
    \item Simulation: forms leave \hl{\ReplayDupConfirmedPerSessionMean} duplicate confirmed seats per workshop, one queue \ReplayDupConfirmedUnifiedMean
    \item Eventbrite relays a ticket, but per date; no tool ties sessions to one queue place
    \item First users: NUS student organisations, starting with our pilot organiser
  \end{itemize}
  \source{Competitor scan from vendor documentation (Luma, Eventbrite, Humanitix, SignUpGenius,
    Peatix, forms + spreadsheet): Final report Table 2. Simulation: \ReplaySeeds{} workshops,
    \code{results/replay/}.}
\end{frame}

\begin{frame}{Request path: 202 at once, then poll until settled}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{6.0in}
      \adjustbox{trim=0pt 0pt {\width-7.55cm} 0pt, clip, width=5.7in}{\input{figures/fig-sequence}}
    \end{column}
    \begin{column}{5.9in}
      \vspace{0.30in}
      \begin{itemize}
        \setlength{\itemsep}{0.45em}
        \item 202 once the command is recorded
        \item Browser polls: 0.3, 0.5, 1 s, then every 2 s
        \item The answer carries the new status
        \item FIFO group = workshop: a single writer
        \item One transaction per decision (C4)
        \item Version guard: a repeat is a no-op (C3)
      \end{itemize}
    \end{column}
  \end{columns}
  \source{Smoke run on dev (n = \SmokeSettleClientN, one run): server settle p50 \SmokeSettleServerPFiftyMs\,ms,
    client-observed p50 \SmokeSettleClientPFiftyMs\,ms. Final report Fig.~3a, \S3.2, \S3.4;
    \code{results/2026-10-17-smoke/}.}
\end{frame}

\begin{frame}{Timers: one per offer; the sweep is only a safety net}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{6.0in}
      \adjustbox{trim={7.45cm} 0pt 0pt 0pt, clip, width=5.7in}{\input{figures/fig-sequence}}
    \end{column}
    \begin{column}{5.9in}
      \vspace{0.30in}
      \begin{itemize}
        \setlength{\itemsep}{0.45em}
        \item One SQS delay timer per offer
        \item Forwarder puts the tick in the FIFO group
        \item Tick-first: expire what is due, then relay
        \item Late or duplicate tick: a no-op (C5)
        \item Cancel or decline: relayed at once
        \item 5-min sweep: a counted safety net
      \end{itemize}
    \end{column}
  \end{columns}
  \source{SQS message timers are at most 900\,s; longer holds hop through the forwarder. Final report
    Fig.~3b and \S3.4 (response to the TA's feedback 2).}
\end{frame}

\begin{frame}{Backup: failure modes and what happens}

  \fm{Worker crash before commit}{nothing written; redelivered after 120\,s}
  \fm{Worker crash after commit}{redelivery finds the command settled: no-op}

  \fm{Poison message}{after 3 receives to the DLQ, alarm; the group moves on}
